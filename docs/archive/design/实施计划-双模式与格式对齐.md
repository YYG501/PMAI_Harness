from pathlib import Path
from typing import Literal, Optional

def detect_format(pm_view: Path) -> Literal["v1", "v2"]:
    """以 .engineering.md 是否存在作为主信号"""
    eng = pm_view.with_suffix("").with_suffix(".engineering.md")
    return "v2" if eng.exists() else "v1"

def get_engineering_path(pm_view: Path) -> Optional[Path]:
    """v2 时返回工程合同路径；v1 返回 None"""

def get_task_status(pm_view: Path) -> str:
    """解析新表格 + 旧 **状态：** 双兼容"""

def get_task_branch(pm_view: Path) -> str: ...
def get_task_worktree(pm_view: Path) -> str: ...
def get_task_meta(pm_view: Path) -> dict:
    """一次性返回 status / branch / worktree / created_at / 等所有字段"""

def read_section(pm_view: Path, section_name: str) -> tuple[bool, Optional[str]]:
    """跨文件查找 section。

SYNC_WHITELIST_PATTERNS = [
    r"^requirements/active/[^/]+/",
    r"^DESIGN\.md$",
    r"^CLAUDE\.md$",
    r"^component-registry\.md$",
]

def check_no_doc_commits(diff_files: list[str]) -> None:
    """task 不应 commit sync 白名单路径下的改动"""
    bad = [f for f in diff_files if any(re.match(p, f) for p in SYNC_WHITELIST_PATTERNS)]
    if bad:
        print(f"Error: task 不允许 commit 以下文档（应在 req worktree 改）：\n  " +
              "\n  ".join(bad), file=sys.stderr)
        sys.exit(1)

## 风险与回退

### 阶段间独立 commit 的好处

每阶段独立 commit，回退粒度清晰。最坏情况：单阶段出问题，`git revert` 回退该 commit，前面阶段的成果保留。

### 高风险阶段

- **阶段 1**：parser 双兼容是新代码，若实现错误会污染下游所有阶段。**对策**：先写完整测试用例（11 条最少），再写实现；用 fixture v1/v2 双轨测试。
- **阶段 7（块 D）**：组件复用判定的启发式（difflib 相似度）容易误判。**对策**：阻断时给 PM "强制新建"的逃生舱（只此一处例外，因为这是启发式判断不是硬规则）。

### 紧急回退

如果改造期间生产 task 闭环出问题：

```bash
# 回退到改造前
git log --oneline | grep -E "(parser|sync-req-docs|mode|component-registry|spec.*双段落|task-plan.*双规则|evidence)" | head -20
# 找到要回退的 commit
git revert <commit-sha>
```
